# Commit 1c — byte-equivalence guard. The SoC adas_msgs (this directory) and
# the PC bridge copy at adas_bridge_pc/carla_ros2_bridge/ws/src/adas_msgs must
# stay identical so that message bindings generated by either workspace agree.
# The check runs under `colcon test --packages-select adas_msgs` and emits a
# diff on stderr if the trees diverge. Run manually with:
#   python3 scripts/check_msgs_mirror.py
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(
    NAME check_msgs_mirror
    COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/scripts/check_msgs_mirror.py
            --soc ${CMAKE_CURRENT_SOURCE_DIR}
            --bridge ${CMAKE_CURRENT_SOURCE_DIR}/../../../../adas_bridge_pc/carla_ros2_bridge/ws/src/adas_msgs
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  )
endif()

ament_package()
